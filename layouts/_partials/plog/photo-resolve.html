{{- /* 解析一张照片的地址与尺寸（photo-card 与 list 的灯箱 JSON 共用）。 入参：dict "page" 组页（提供页面资源） "photo" 照片 map（src 必填）。 返回 map： thumb/tw/th = 列表卡缩略图（x900，列宽约 1/3 屏 + Retina 余量） large/w/h = 灯箱大图（Fit 1600x1600，只缩不放） src = 原始文件 URL SVG/GIF 不做缩放，直接用原图、无宽高（由 CSS/浏览器自适应）。 */ -}} {{- $page := .page -}} {{- $photo := .photo -}} {{- $src := "" -}} {{- with index $photo "src" }}{{ $src = . | string }}{{ end -}} {{- if not $src }}{{ errorf "plog %s: photos 项缺少 src" $page.Path }}{{ end -}} {{- $thumbSrc := $src -}}{{- $tw := "" -}}{{- $th := "" -}} {{- $largeSrc := $src -}}{{- $w := "" -}}{{- $h := "" -}} {{- with $page.Resources.Get $src -}} {{- if not (in (slice "svg" "gif") .MediaType.SubType) -}} {{- with .Resize "x900" -}} {{- $thumbSrc = .RelPermalink -}}{{- $tw = .Width -}}{{- $th = .Height -}} {{- end -}} {{- with .Fit "1600x1600" -}} {{- $largeSrc = .RelPermalink -}}{{- $w = .Width -}}{{- $h = .Height -}} {{- end -}} {{- else -}} {{- $thumbSrc = .RelPermalink -}}{{- $largeSrc = .RelPermalink -}} {{- end -}} {{- end -}} {{- return (dict "src" $src "thumb" $thumbSrc "tw" $tw "th" $th "large" $largeSrc "w" $w "h" $h) -}}
