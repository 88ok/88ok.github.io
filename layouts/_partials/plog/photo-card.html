{{- /* 单张照片卡（时光相册样式）。 入参：dict "page" 页面（提供资源与徽章文案） "photo" 一张照片 （map：src 必填，caption 可选）。 结构：圆角照片卡 = img + 左上角徽章 + 可选左下角图注； data-td-image-zoom 指向原图，由主题 image-zoom JS 弹出大图。 位图生成 x900 缩略图（列宽约 1/3 屏，Retina 余量），SVG/GIF 不做处理直接用原图；无宽高属性时由 img 自身比例撑起。 */ -}} {{- $page := .page -}} {{- $photo := .photo -}} {{- $src := "" -}} {{- with index $photo "src" }}{{ $src = . | string }}{{ end -}} {{- if not $src }}{{ errorf "plog %s: photos 项缺少 src" $page.Path }}{{ end -}} {{- $caption := "" -}} {{- with index $photo "caption" }}{{ $caption = . | string }}{{ end -}} {{- $imgSrc := $src -}} {{- $fullSrc := $src -}} {{- $w := "" -}} {{- $h := "" -}} {{- with $page.Resources.Get $src -}} {{- $fullSrc = .RelPermalink -}} {{- if not (in (slice "svg" "gif") .MediaType.SubType) -}} {{- $thumb := .Resize "x900" -}} {{- $imgSrc = $thumb.RelPermalink -}} {{- $w = $thumb.Width -}} {{- $h = $thumb.Height -}} {{- else -}} {{- $imgSrc = .RelPermalink -}} {{- end -}} {{- end -}}
